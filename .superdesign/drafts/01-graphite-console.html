package lab.recruit.auth;
import org.springframework.http.*;
import org.springframework.web.bind.*;
import jakarta.validation.Valid;
// 密码比对交给编码器，别自己写 ==
@RestController
@RequestMapping("/api/auth")
public class AuthController {
private final TokenService tokens;
private final UserRepository users;
public AuthController(
TokenService t,
UserRepository u) {
this.tokens = t;
this.users = u;
}
@PostMapping("/login")
public ResponseEntity<Token> login(
@Valid @RequestBody Form f) {
return users
.findByStudentId(f.id())
.filter(u -> ok(u, f))
.map(tokens::issue)
.map(ResponseEntity::ok)
.orElseGet(this::no);
}
private boolean ok(User u, Form f) {
if (u.locked()) {
log.warn("locked");
return false;
}
return encoder()
.matches(f.pw(), u.hash());
}
private ResponseEntity<Token> no() {
return ResponseEntity
.status(401)
.build();
}
}

#include <stdint.h>
#include <string.h>
#include "flash.h"
/* 寄存器基址，手册 7.3 节 */
#define PERIPH_BASE 0x40000000u
#define FLASH_SR (PERIPH_BASE + 0x10u)
#define FLASH_CR (PERIPH_BASE + 0x14u)
#define R(o) (*(volatile uint32_t*)o)
#define BSY (1u << 0)
#define PG (1u << 3)
static int wait_ready(uint32_t ticks)
{
uint32_t t = 0;
while (R(FLASH_SR) & BSY) {
if (++t > ticks) {
return -ETIMEDOUT;
}
__NOP();
}
return 0;
}
/* len 是外面传进来的，先夹住再搬 */
int fw_load(uint8_t *dst,
const uint8_t *src,
size_t len, size_t cap)
{
if (dst == NULL || src == NULL) {
return -EINVAL;
}
if (len == 0u || len > cap) {
return -EINVAL;
}
if (len % 4u != 0u) {
return -EINVAL;
}
if (wait_ready(1000u) != 0) {
return -EBUSY;
}
R(FLASH_CR) |= PG;
size_t i;
for (i = 0; i < len; i += 4u) {
memcpy(dst + i, src + i, 4u);
if (wait_ready(50u) != 0) {
R(FLASH_CR) &= ~PG;
return -EBUSY;
}
}
R(FLASH_CR) &= ~PG;
return (int)len;
}
